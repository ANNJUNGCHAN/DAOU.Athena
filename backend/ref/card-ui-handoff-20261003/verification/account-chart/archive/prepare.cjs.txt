function replaceOnce(source, before, after) {
  const normalized = source.replace(/\r\n/g, '\n');
  before = before.replace(/\r\n/g, '\n');
  if (normalized.split(before).length !== 2) throw new Error('Nonunique source patch: ' + before.slice(0,70));
  const start = normalized.indexOf(before), end = start + before.length;
  function offset(target) {
    let original=0, logical=0;
    while(logical < target) { if(source[original]==='\r'&&source[original+1]==='\n')original++; original++; logical++; }
    return original;
  }
  const a=offset(start); let b=offset(end);
  let ending='';
  if(source.slice(b,b+2)==='\r\n'){ b+=2; ending='\n'; }
  else if(source[b]==='\n'){ b++; ending='\n'; }
  return source.slice(0,a)+after.replace(/\r\n/g,'\n')+ending+source.slice(b);
}
